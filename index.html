Ex 1 التمرين 1 — الفروق المنتهية (Différences finies)

حل معادلة الانتشار مع شروط حدودية صفرية باستعمال Crank–Nicolson ونظام ثلاثي قطري.

  • المعادلة: ∂tu(x,t) − μ ∂xxu(x,t) = f(x,t)
  • المجال: 0<x<1 و 0<t<T
  • الشروط الحدّية: u(0,t)=0 و u(1,t)=0
  • الشرط الابتدائي: u(x,0)=u0(x)
التقسيم
  • h = 1/N ، xi = i h
  • Δt = T/m ، tn = n Δt
  • uxx(xi) ≈ (ui+1 − 2ui + ui−1) / h²
Crank–Nicolson
  • r = μΔt / (2h²)
  • نحل في كل خطوة: A u^{n+1} = B u^{n} + Δt f

دالة MATLAB — الحل بـ A\\rhs

function u = solveCN(N,m,mu,T)
h  = 1/N;
dt = T/m;

x = (0:h:1)';           % نقاط الشبكة
u = log(1 + x);         % u0(x) = ln(1+x)
u([1 end]) = 0;         % شروط حدّية

r = mu*dt/(2*h^2);

% مصفوفة A (زمن n+1)
mainA = (1+2*r)*ones(N-1,1);
offA  = (-r)*ones(N-2,1);
A = diag(mainA,0) + diag(offA,1) + diag(offA,-1);

% مصفوفة B (زمن n)
mainB = (1-2*r)*ones(N-1,1);
offB  = ( r)*ones(N-2,1);
B = diag(mainB,0) + diag(offB,1) + diag(offB,-1);

for n = 1:m
    f = 2*x(2:N) + 3;          % f(x,t)=2x+3
    rhs = B*u(2:N) + dt*f;
    u(2:N) = A \\ rhs;
end
end

Thomas — حل النظام الثلاثي القطري + solveCN_Thomas

function y = thomas(a,b,c,d)
% a: تحت القطر (n-1), b: القطر الرئيسي (n), c: فوق القطر (n-1)
n = length(b);
cp = zeros(n-1,1); dp = zeros(n,1);

cp(1) = c(1)/b(1);
dp(1) = d(1)/b(1);

for i=2:n-1
    denom = b(i) - a(i-1)*cp(i-1);
    cp(i) = c(i)/denom;
    dp(i) = (d(i) - a(i-1)*dp(i-1))/denom;
end

dp(n) = (d(n) - a(n-1)*dp(n-1)) / (b(n) - a(n-1)*cp(n-1));

y = zeros(n,1);
y(n) = dp(n);
for i=n-1:-1:1
    y(i) = dp(i) - cp(i)*y(i+1);
end
end
function u = solveCN_Thomas(N,m,mu,T)
h=1/N; dt=T/m;
x=(0:h:1)';
u=log(1+x); u([1 end])=0;

r = mu*dt/(2*h^2);

% معاملات A
a = (-r)*ones(N-2,1);
b = (1+2*r)*ones(N-1,1);
c = (-r)*ones(N-2,1);

% معاملات B
aB = ( r)*ones(N-2,1);
bB = (1-2*r)*ones(N-1,1);
cB = ( r)*ones(N-2,1);

for n=1:m
    f = 2*x(2:N)+3;
    rhs = bB.*u(2:N);
    rhs(2:end) = rhs(2:end) + aB.*u(2:N-1);
    rhs(1:end-1) = rhs(1:end-1) + cB.*u(3:N);
    rhs = rhs + dt*f;

    u(2:N) = thomas(a,b,c,rhs);
end
end

Ex 2 التمرين 2 — Darcy بالعناصر المنتهية (FreeFem++)

تقريب مختلط: RT0 للسرعة و P0 للضغط، مع تثبيت الضغط بـ δ صغيرة.

  • u + ∇p = f
  • div(u) = 0
  • شرط حدودي: u·n = 0 على ∂Ω
  • المعطيات: n=30 ، δ=5×10⁻⁶ ، f(x,y)=sin(πx)sin(πy)
// Darcy_RT0_P0.edp
real delta = 5e-6;
int n = 30;

mesh Th = square(n,n);

fespace Vh(Th,RT0);
fespace Ph(Th,P0);

Vh [u1,u2],[v1,v2];
Ph p,q;

func fx = sin(pi*x)*sin(pi*y);
func fy = sin(pi*x)*sin(pi*y);

solve Darcy([u1,u2,p],[v1,v2,q])
= int2d(Th)( u1*v1 + u2*v2
           - p*(dx(v1)+dy(v2))
           + q*(dx(u1)+dy(u2))
           + delta*p*q )
- int2d(Th)( fx*v1 + fy*v2 );

plot([u1,u2], value=true, wait=true, cmm="u (velocity)");
plot(p, value=true, wait=true, cmm="p (pressure)");
ملاحظة: δ*p*q مهم لأن p عادةً محدد حتى ثابت فقط.

Ex 3 التمرين 3 — بناء المجال Ω وإنشاء الشبكة (FreeFem++)

قالب جاهز: buildmesh مع 40 نقطة لكل حد + حفظ .msh و .eps.

// DomainMesh.edp
int nb = 40;

// Γ1: حد خارجي (مثال دائرة)
border G1(t=0,2*pi){ x = 5*cos(t); y = 5*sin(t); label=1; }

// Γ3: ثقب مستطيل (مثال: من (-3,-3) إلى (-1,3))
border R1(t=0,1){ x=-3; y=-3 + 6*t; label=3; }
border R2(t=0,1){ x=-3 + 2*t; y= 3; label=3; }
border R3(t=0,1){ x=-1; y= 3 - 6*t; label=3; }
border R4(t=0,1){ x=-1 - 2*t; y=-3; label=3; }

// Γ2: ثقب منحني (مثال بيضوي) — عدّل حسب الرسم
border G2(t=0,2*pi){ x = 2 + 1.0*cos(t); y = 1.0*sin(t); label=2; }

// الثقوب عادةً لازم تكون بعكس اتجاه الحد الخارجي (استعمل -nb)
mesh Th = buildmesh( G1(nb) + R1(-nb)+R2(-nb)+R3(-nb)+R4(-nb) + G2(-nb) );

plot(Th, wait=true, cmm="Mesh of Omega");
savemesh(Th, "Omega.msh");
saveeps("Omega.eps", Th);
اختبار سريع: إذا الشبكة دخلت داخل الثقب، بدّل اتجاه حدود الثقب (nb ↔ -nb).
